Dates du plus ancien et du plus récent */ -}} {{- $firstPost := index $sortedPosts 0 -}} {{- $lastPost := index $sortedPosts (sub $count 1) -}} {{- $firstDate := $firstPost.Date -}} {{- $lastDate := $lastPost.Date -}} {{- /* 3. Calculs de durée (en secondes, puis converti en jours) */ -}} {{- /* On compare avec la date du dernier billet (ou "now" selon ta préférence) */ -}} {{- $elapsedSeconds := sub $lastDate.Unix $firstDate.Unix -}} {{- $elapsedDays := math.Max 1 (math.Round (div (float $elapsedSeconds) 86400)) -}} {{- /* Date limite du défi : +365 jours (31 536 000 secondes) après le 1er billet */ -}} {{- $deadlineUnix := add $firstDate.Unix 31536000 -}} {{- $deadlineDate := time.AsTime $deadlineUnix -}} {{- /* Jours restants avant la date limite */ -}} {{- $remainingSeconds := sub $deadlineUnix now.Unix -}} {{- $remainingDays := math.Max 1 (math.Round (div (float $remainingSeconds) 86400)) -}} {{- /* 4. Variables calculées demandées */ -}} {{- /* B billets en J jours */ -}} {{- $billetsPublies := $count -}} {{- $joursEcoules := int $elapsedDays -}} {{- /* Rythme actuel : 1 billet tous les X jours / Y billets par jour */ -}} {{- $joursParBillet := div (float $elapsedDays) (float $count) -}} {{- $billetsParJour := div (float $count) (float $elapsedDays) -}} {{- /* Billets manquants */ -}} {{- $manquants := math.Max 0 (sub $goal $count) -}} {{- /* Rythme requis pour finir dans les temps */ -}} {{- $requisJoursParBillet := 0.0 -}} {{- $requisBilletsParJour := 0.0 -}} {{- if gt $manquants 0 -}} {{- $requisJoursParBillet = div (float $remainingDays) (float $manquants) -}} {{- $requisBilletsParJour = div (float $manquants) (float $remainingDays) -}} {{- end -}} {{- /* Date de fin estimée au rythme actuel */ -}} {{- $joursRestantsEstimes := mul (float $manquants) $joursParBillet -}} {{- $secondesRestantesEstimees := int (mul $joursRestantsEstimes 86400) -}} {{- $projectedEndUnix := add now.Unix $secondesRestantesEstimees -}} {{- $projectedEndDate := time.AsTime $projectedEndUnix -}} {{- /* 5. Affichage HTML */ -}}
